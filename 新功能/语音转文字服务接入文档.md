# 语音转文字服务 (Speech Service) 接入文档

本文档描述了如何接入基于 WebSocket 的实时语音转文字服务。

## 1. 服务信息

*   **协议**: WebSocket
*   **路径**: `/speech`
*   **鉴权**: 需要有效的 JWT 登录 Token

## 2. 连接与鉴权

在建立 WebSocket 连接时，必须通过 `auth` 对象或查询参数 `query` 传递 Token。

### 连接参数

| 参数名 | 类型 | 必填 | 说明 | 示例 |
| :--- | :--- | :--- | :--- | :--- |
| `token` | String | 是 | 登录接口返回的 JWT Token | `eyJhbGciOiJ...` |

### 示例代码 (Socket.io Client)

```javascript
import { io } from "socket.io-client";

const socket = io('http://localhost:3000/speech', {
  auth: {
    token: 'YOUR_JWT_TOKEN' // 从登录接口获取
  },
  transports: ['websocket']
});

socket.on('connect', () => {
  console.log('已连接到语音服务');
});

socket.on('connect_error', (err) => {
  console.error('连接失败:', err.message);
});
```

## 3. 发送音频数据

连接建立后，客户端通过发送 `audio-stream` 事件上传音频数据。

*   **事件名**: `audio-stream`
*   **数据格式**: 二进制数据 (ArrayBuffer / Buffer)
*   **音频要求**:
    *   **采样率**: 16000 Hz (必须)
    *   **声道**: 单声道 (Mono)
    *   **位深**: 16-bit PCM (Little Endian)

> **注意**: 浏览器录音通常默认为 44.1kHz 或 48kHz，发送前**必须**在前端进行重采样（Downsampling）和格式转换。

### 示例代码 (发送音频)

```javascript
// 假设 buffer 是转换好的 16k 16bit PCM 数据
socket.emit('audio-stream', buffer);
```

## 4. 接收识别结果

服务端会通过以下两个事件实时推送识别结果：

### 4.1 完整结果 (`recognition-result`)

当识别出一句完整的通过话时触发。

*   **事件名**: `recognition-result`
*   **数据结构**:

```json
{
  "text": "你好 世界"
}
```

### 4.2 部分结果 (`recognition-partial`)

当正在说话且尚未断句时，实时返回中间识别结果。

*   **事件名**: `recognition-partial`
*   **数据结构**:

```json
{
  "partial": "你好 世"
}
```

## 5. 停止识别

客户端可以发送 `stop` 事件来强制结束当前识别并获取最后的结果（通常在停止录音时调用）。

*   **事件名**: `stop`
*   **参数**: 无

```javascript
socket.emit('stop');
```

## 6. 前端完整接入示例 (React/HTML)

以下是一个包含麦克风录音、音频转换（重采样 + PCM 转换）和 WebSocket 通信的完整示例。

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <title>语音转文字测试</title>
    <script src="https://cdn.socket.io/4.7.4/socket.io.min.js"></script>
</head>
<body>
    <button id="startBtn">开始录音</button>
    <button id="stopBtn" disabled>停止录音</button>
    <div id="result" style="white-space: pre-wrap; margin-top: 20px;"></div>

    <script>
        let socket;
        let audioContext;
        let processor;
        let input;
        let globalStream;
        let finalTranscript = '';

        const TOKEN = 'YOUR_TOKEN_HERE'; // 替换为真实 Token

        // 1. 初始化 Socket 连接
        function initSocket() {
            socket = io('http://localhost:3000/speech', {
                auth: { token: TOKEN },
                transports: ['websocket']
            });

            socket.on('connect', () => console.log('Connected'));
            
            // 接收完整结果
            socket.on('recognition-result', (data) => {
                if (data.text) {
                    finalTranscript += data.text + ' ';
                    document.getElementById('result').textContent = finalTranscript;
                }
            });

            // 接收中间结果
            socket.on('recognition-partial', (data) => {
                if (data.partial) {
                    document.getElementById('result').textContent = finalTranscript + data.partial;
                }
            });
        }

        // 2. 音频处理工具：Float32 -> Int16 PCM
        function floatTo16BitPCM(output, offset, input) {
            for (let i = 0; i < input.length; i++, offset += 2) {
                let s = Math.max(-1, Math.min(1, input[i]));
                s = s < 0 ? s * 0x8000 : s * 0x7FFF;
                output.setInt16(offset, s, true);
            }
        }

        // 3. 音频处理工具：降采样 (44.1k/48k -> 16k)
        function downsampleBuffer(buffer, sampleRate, outSampleRate) {
            if (outSampleRate == sampleRate) return buffer;
            if (outSampleRate > sampleRate) throw "rate error";
            const sampleRateRatio = sampleRate / outSampleRate;
            const newLength = Math.round(buffer.length / sampleRateRatio);
            const result = new Float32Array(newLength);
            let offsetResult = 0;
            let offsetBuffer = 0;
            while (offsetResult < result.length) {
                const nextOffsetBuffer = Math.round((offsetResult + 1) * sampleRateRatio);
                let accum = 0, count = 0;
                for (let i = offsetBuffer; i < nextOffsetBuffer && i < buffer.length; i++) {
                    accum += buffer[i];
                    count++;
                }
                result[offsetResult] = accum / count;
                offsetResult++;
                offsetBuffer = nextOffsetBuffer;
            }
            return result;
        }

        // 4. 开始录音
        document.getElementById('startBtn').onclick = async () => {
            initSocket();
            document.getElementById('startBtn').disabled = true;
            document.getElementById('stopBtn').disabled = false;

            try {
                globalStream = await navigator.mediaDevices.getUserMedia({ audio: true });
                audioContext = new AudioContext();
                input = audioContext.createMediaStreamSource(globalStream);
                
                // 创建处理器：缓冲区大小 4096
                processor = audioContext.createScriptProcessor(4096, 1, 1);

                input.connect(processor);
                processor.connect(audioContext.destination);

                processor.onaudioprocess = (e) => {
                    if (!socket || !socket.connected) return;

                    const left = e.inputBuffer.getChannelData(0);
                    // 降采样
                    const downsampled = downsampleBuffer(left, audioContext.sampleRate, 16000);
                    // 转 PCM
                    const buffer = new ArrayBuffer(downsampled.length * 2);
                    const view = new DataView(buffer);
                    floatTo16BitPCM(view, 0, downsampled);
                    
                    // 发送
                    socket.emit('audio-stream', buffer);
                };
            } catch (err) {
                console.error(err);
                alert('录音启动失败');
            }
        };

        // 5. 停止录音
        document.getElementById('stopBtn').onclick = () => {
            if (socket) {
                socket.emit('stop');
                socket.disconnect();
            }
            if (processor) {
                input.disconnect(processor);
                processor.disconnect(audioContext.destination);
            }
            if (audioContext) audioContext.close();
            if (globalStream) globalStream.getTracks().forEach(t => t.stop());
            
            document.getElementById('startBtn').disabled = false;
            document.getElementById('stopBtn').disabled = true;
        };
    </script>
</body>
</html>
```
